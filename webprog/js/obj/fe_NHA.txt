const diak = {
    nev: "Sanyi",
    jegyek: [5, 2, 3, 4, 5, 3],
    atlag: function ()
    {
        let sum = 0;
        this.jegyek.forEach(element =>
        {
            sum += element;
        });
        return sum / this.jegyek.length;
    }
}
